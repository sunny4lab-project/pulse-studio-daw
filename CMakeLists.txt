cmake_minimum_required(VERSION 3.22)
project(PulseStudio VERSION 0.20.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.11 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_gui_app(PulseStudio PRODUCT_NAME "Pulse Studio")
target_sources(PulseStudio PRIVATE Source/Main.cpp)
target_compile_definitions(PulseStudio PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_PLUGINHOST_VST3=1)
if(APPLE)
 target_compile_definitions(PulseStudio PRIVATE JUCE_PLUGINHOST_AU=1)
endif()
# JUCE 8.0.11 provides the ASIO headers; no manual SDK installation is needed.
if(WIN32)
 target_compile_definitions(PulseStudio PRIVATE JUCE_ASIO=1)
 if(DEFINED ASIO_SDK_DIR)
  if(NOT EXISTS "${ASIO_SDK_DIR}/common/iasiodrv.h")
   message(FATAL_ERROR "Invalid ASIO_SDK_DIR: common/iasiodrv.h is missing")
  endif()
  target_compile_definitions(PulseStudio PRIVATE JUCE_ASIO_USE_EXTERNAL_SDK=1)
  target_include_directories(PulseStudio PRIVATE "${ASIO_SDK_DIR}/common" "${ASIO_SDK_DIR}/host" "${ASIO_SDK_DIR}/host/pc")
 endif()
 message(STATUS "Pulse Studio Windows: ASIO driver backend ENABLED")
endif()
target_link_libraries(PulseStudio PRIVATE juce::juce_audio_utils juce::juce_audio_formats juce::juce_gui_extra juce::juce_recommended_config_flags)
include(CTest)
if(BUILD_TESTING)
 add_executable(pulse_dsp_tests tests/dsp.cpp)
 target_include_directories(pulse_dsp_tests PRIVATE Source)
 add_test(NAME pulse_dsp_tests COMMAND pulse_dsp_tests)
endif()
